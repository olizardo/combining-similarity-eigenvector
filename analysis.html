Main Function

    prestige.sim <- function(x, a = 0, b = 0) {
        z <- as.matrix(as_adjacency_matrix(x))
        p <- z %*% z
        q <- z %*% (1 - z)  
        j <- p / (p + q + t(q))
        d <- 1 - j
        w <- z * (j^a * d^b)
        s <- abs(eigen(w)$vectors[, 1])
        s <- (s - min(s))/(max(s) - min(s)) #normalizing
        names(s) <- V(x)$name
        return(round(s, 4))
        }
    prestige.sim.vec <- Vectorize(prestige.sim) #vectorized version of the function

Main Table

    bon <- prestige.sim(g, a = 0, b = 0)
    ace <- prestige.sim(g, a = 0, b = 1)
    ol1 <- prestige.sim(g, a = 1, b = 0)
    ol2 <- prestige.sim(g, a = 0.15, b = 0.85)
    sort.vec <- function(x) {
        x <- data.frame(Node = names(x), Score = as.numeric(x))
        x <- x[order(x$Score, decreasing = TRUE), ]
        return(x)
        }
    eig.dat <- data.frame(sort.vec(bon), sort.vec(ace), sort.vec(ol1), sort.vec(ol2))
    save_kable(

Parameter Range Sensitivity Analysis

    beta.range <- seq(0, 1, by = 0.05) #sequence of beta values
    c.dat <- data.frame(prestige.sim.vec(list(g), a = 1-beta.range, b = beta.range)) |> 
        mutate(family = V(g)$name) #creating data frame from vectorized functionn
    names(c.dat)[1:ncol(c.dat)-1] <- paste("b", sprintf("%.2f", seq(0, 1, by = 0.05)), sep = "_") #naming columns
    c.long <-  c.dat %>% 
         pivot_longer(
            cols = !family,
            names_to = "Beta",
            values_to = "value"
          ) |> 
        mutate(Beta = factor(Beta, ordered = TRUE, levels = colnames(c.dat[-ncol(c.dat)]))) #creating long data frame
    ## plotting
    p <- ggplot(
        data = c.long, 
        mapping = aes(x = Beta, y = value, color = family, group = family)
        )
    p <- p + geom_vline(
        aes(xintercept = which(sprintf("%.2f", seq(0, 1, by = 0.05)) == "0.85")), 
        linetype = 2, color = "red"
        )
    p <- p + geom_bump(linewidth = 0.8, smooth = 2)
    p <- p + geom_point(size = 3)
    p <- p + geom_text_repel(
        data = filter(c.long, 
        Beta == colnames(c.dat)[1]),
        aes(x = Beta, label = family), 
        size = 6, nudge_x = -3.5
        ) #left labels
    p <- p + geom_text_repel(
        data = filter(c.long, 
        Beta == colnames(c.dat)[ncol(c.dat)-1]),
        aes(x = Beta, label = family), 
        size = 6, nudge_x = 3.5
        ) #right labels
    p <- p + theme_minimal()
    p <- p + theme(
        legend.position = "none",
        text = element_text(size = 24)
        )
    p <- p + scale_x_discrete(
        breaks =  unique(c.long$Beta)[seq(1, 21, by = 2)],
        labels = sprintf("%.2f", seq(0, 1, by = 0.05))[seq(1, 21, by = 2)]
        )
    p <- p + xlab("Beta") + ylab("Eigenvector Score")
    p
