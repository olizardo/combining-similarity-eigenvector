   prestige.sim <- function(x, alpha = 0, beta = 0) {
      A <- as.matrix(as_adjacency_matrix(x))
      A.p <- A %*% A 
      A.q <- A %*% (1 - A)  
      S <- A.p / (A.p + A.q + t(A.q))
      D <- 1 - S
      W <- A * (S^alpha * D^beta)
      s <- abs(eigen(W)$vectors[, 1])
      names(s) <- rownames(A)
      s <- (s - min(s))/(max(s) - min(s))
      return(list(s = s, D = D, S = S, W = W))
   }
